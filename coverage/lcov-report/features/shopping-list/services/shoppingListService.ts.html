  const { data, error } = await supabase

    .eq('user_id', user.id)
    .limit(500); // Limitar consulta inicial
 
  if (error || !data) {
    console.error('Error fetching items for frequency count:', error);
    return [];
  }
 
  // Contar frecuencias en cliente
  const frequencyMap: { [name: string]: number } = {};
  data.forEach(item => {
    if (item.name) {
      frequencyMap[item.name] = (frequencyMap[item.name] || 0) + 1;
    }
  });
 
  // Ordenar y devolver nombres
  const sortedItems = Object.entries(frequencyMap)
    .sort(([, countA], [, countB]) => countB - countA)
    .slice(0, limit)
    .map(([name]) => name);
 
  return sortedItems;
};

const calculateMissingRecipeIngredients = async (recipeIngredients: RecipeIngredient[]): Promise<NewShoppingListItem[]> => { // Quitar export

    const pantryMap = new Map<string, number>();
    pantryItems.forEach((item) => {
        // Usar el nombre del ingrediente
        const name = item.ingredient?.name;
        if (name) {
            // Los campos quantity y unit son opcionales, usar null si no existen
            const normalizedUnit = normalizeUnit(item.unit ?? null);
            const quantity = parseQuantity(item.quantity ?? null);
            // Solo añadir si tenemos nombre válido
            if (normalizedUnit !== null && quantity !== null && quantity > 0) {
                const key = `${name.trim().toLowerCase()}-${normalizedUnit}`;
                pantryMap.set(key, (pantryMap.get(key) || 0) + quantity);
            }
            // Considerar ítems sin unidad (contarlos por nombre)
            else if (normalizedUnit === null && quantity !== null && quantity > 0) {
                 const key = `${name.trim().toLowerCase()}-null`; // Clave especial para sin unidad

        if (requiredQuantity === null || requiredQuantity <= 0) {

             if (!missingItems.some(item => item.name.toLowerCase() === name.trim().toLowerCase())) {
                 missingItems.push({
                     name: name.trim(),

                 });
             }

        // Construir clave para buscar en despensa

        const availableQuantity = pantryMap.get(key) || 0;
 
        const neededQuantity = requiredQuantity - availableQuantity;
 
        if (neededQuantity > 0) {
             // Evitar añadir duplicados
             if (!missingItems.some(missingItem => missingItem.name.toLowerCase() === name.trim().toLowerCase() && normalizeUnit(missingItem.unit) === requiredUnit)) {

                     quantity: neededQuantity, // Añadir solo la cantidad faltante
                     unit: ingredient.unit, // Mantener la unidad original de la receta para la lista
                     // category_id: ingredient.category_id,
                 });
             }
        }
    });
 
    return missingItems;
};

    const { error: deleteError } = await supabase.rpc('delete_unpurchased_items', {
        p_user_id: userId
    });
 
    if (deleteError) {
        console.error("Error eliminando items antiguos de la lista de compras:", deleteError);
        throw new Error("Error al limpiar la lista de compras anterior.");
    }
    console.log(`Items no comprados eliminados para usuario ${userId}.`);
 
    // 2. Preparar nuevos items para insertar
    if (generatedItems.length === 0) {
        console.log("No hay nuevos items generados para insertar.");
        return; // No hay nada que insertar
    }
 
    // Asegurarse de que el tipo coincida con la tabla (usando tipos generados si es posible)
    const itemsToInsert = generatedItems.map(item => ({
        user_id: userId,
        name: item.name, // Columna 'name' en la tabla
        quantity: item.totalQuantity,
        unit: item.unit,
        is_purchased: false, // Usar nombre de columna correcto
        // recipe_source: item.recipeSources.join(', ') || null, // Asumiendo que esta columna existe
    }));
 
    // 3. Insertar nuevos items
    const { error: insertError } = await supabase

}
